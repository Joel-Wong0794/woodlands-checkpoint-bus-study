# Install on the VM with:  crontab deploy/crontab.txt
# Stage 2: poll both stops every minute. flock stops two runs overlapping.
* * * * * cd $HOME/woodlands-checkpoint-bus-study && set -a && . ./.env && set +a && flock -n /tmp/collector.lock .venv/bin/python collector.py >> logs/collector.log 2>&1
# Stage 4: rebuild the dashboard every 15 min and upload it to the public bucket, as causeway-bus-times (the main
# address; no extension, so the type is set explicitly) and as index.html (so older links keep working).
# --gzip-local-all stores the page gzip-compressed (~10x smaller download); browsers unzip it automatically.
# V2: the same run writes data.json (the V2 page's data), uploaded to v2/data.json; then the V2 page itself
# (web/: index.html, app.js, logic.js, app.css, ~40 KB) is synced to v2/ (rsync copies only changed files; rsync has
# no --gzip-local-all, unlike cp, so these few files are stored uncompressed; last in the chain, so a
# failure there never stops V1 or data.json). V2 lives at .../woodlands-checkpoint-bus-dashboard/v2/index.html
*/15 * * * * cd $HOME/woodlands-checkpoint-bus-study && flock -n /tmp/dashboard.lock sh -c '.venv/bin/python dashboard.py && gcloud storage cp dashboard.html gs://woodlands-checkpoint-bus-dashboard/causeway-bus-times --cache-control=no-cache --content-type=text/html --gzip-local-all && gcloud storage cp dashboard.html gs://woodlands-checkpoint-bus-dashboard/index.html --cache-control=no-cache --gzip-local-all && gcloud storage cp data.json gs://woodlands-checkpoint-bus-dashboard/v2/data.json --cache-control=no-cache --content-type=application/json --gzip-local-all && gcloud storage rsync web gs://woodlands-checkpoint-bus-dashboard/v2 --exclude="data\.json$" --cache-control=no-cache' >> logs/dashboard.log 2>&1
# Alert: every 15 min, ping healthchecks.io (HC_URL in .env) only if an LTA call succeeded in the last 10 min.
# No ping = the service emails. Covers a stopped collector and a hung or stopped VM. Skipped while HC_URL is unset.
*/15 * * * * cd $HOME/woodlands-checkpoint-bus-study && set -a && . ./.env && set +a && [ -n "$HC_URL" ] && [ "$(sqlite3 data/causeway.db "SELECT count(*) FROM poll_runs WHERE ok=1 AND polled_at > datetime('now','localtime','-10 minutes')")" -gt 0 ] && curl -fsS -m 10 --retry 3 "$HC_URL" > /dev/null 2>> logs/heartbeat.log
# CD: every 5 min, pull the `deploy` branch (moved by GitHub Actions only after the tests pass).
*/5 * * * * cd $HOME/woodlands-checkpoint-bus-study && flock -n /tmp/deploy.lock sh deploy/update.sh >> logs/deploy.log 2>&1
# Stage 4: weekly backup (Sunday 03:00) to the private bucket.
0 3 * * 0 cd $HOME/woodlands-checkpoint-bus-study && sqlite3 data/causeway.db ".backup data/backup.db" && gcloud storage cp data/backup.db gs://woodlands-checkpoint-bus-backup/causeway-$(date +\%F).db >> logs/backup.log 2>&1
